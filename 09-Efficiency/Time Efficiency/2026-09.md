# Time Efficiency — September 2026

**Required Efficiency** = Worked Hrs ÷ 12 (fixed daily standard)  
**Actual Efficiency** = Worked Hrs ÷ Planned Hrs (against what you planned)

```dataviewjs
const fm = dv.current();
const file = app.vault.getAbstractFileByPath(fm.file.path);
const month = fm.file.name;
const [y, m] = month.split('-').map(Number);
const daysInMonth = new Date(y, m, 0).getDate();

const container = dv.el('div', '', { cls: 'eff-time-wrap' });
const table = document.createElement('table');
table.className = 'eff-table';

const headerRow = document.createElement('tr');
['Date', 'Planned Hrs', 'Worked Hrs', 'Required Eff.', 'Actual Eff.'].forEach(h => {
  const th = document.createElement('th');
  th.textContent = h;
  headerRow.appendChild(th);
});
table.appendChild(headerRow);

for (let d = 1; d <= daysInMonth; d++) {
  const dd = String(d).padStart(2, '0');
  const wd = new Date(y, m - 1, d).toLocaleDateString('default', { weekday: 'short' });
  const row = document.createElement('tr');

  const dateCell = document.createElement('td');
  dateCell.textContent = `${d} ${wd}`;
  row.appendChild(dateCell);

  const plannedCell = document.createElement('td');
  const plannedInput = document.createElement('input');
  plannedInput.type = 'number';
  plannedInput.placeholder = 'hrs';
  plannedInput.value = fm['planned_' + dd] ?? '';
  plannedCell.appendChild(plannedInput);
  row.appendChild(plannedCell);

  const workedCell = document.createElement('td');
  const workedInput = document.createElement('input');
  workedInput.type = 'number';
  workedInput.placeholder = 'hrs';
  workedInput.value = fm['worked_' + dd] ?? '';
  workedCell.appendChild(workedInput);
  row.appendChild(workedCell);

  const reqCell = document.createElement('td');
  reqCell.className = 'eff-value';
  row.appendChild(reqCell);

  const actCell = document.createElement('td');
  actCell.className = 'eff-value';
  row.appendChild(actCell);

  const recompute = () => {
    const w = parseFloat(workedInput.value);
    const p = parseFloat(plannedInput.value);
    reqCell.textContent = isNaN(w) ? '—' : ((w / 12) * 100).toFixed(1) + '%';
    actCell.textContent = (isNaN(w) || isNaN(p) || p === 0) ? '—' : ((w / p) * 100).toFixed(1) + '%';
  };
  recompute();

  const persist = async () => {
    await app.fileManager.processFrontMatter(file, (frontmatter) => {
      frontmatter['planned_' + dd] = plannedInput.value === '' ? null : parseFloat(plannedInput.value);
      frontmatter['worked_' + dd] = workedInput.value === '' ? null : parseFloat(workedInput.value);
    });
  };

  plannedInput.addEventListener('input', recompute);
  workedInput.addEventListener('input', recompute);
  plannedInput.addEventListener('change', persist);
  workedInput.addEventListener('change', persist);

  table.appendChild(row);
}

container.appendChild(table);
```
