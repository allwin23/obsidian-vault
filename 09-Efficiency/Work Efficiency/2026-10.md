## Month Summary

```dataviewjs
const file = dv.current(); const abstractFile = app.vault.getAbstractFileByPath(file.file.path); const content = await app.vault.read(abstractFile); const lines = content.split('\n'); const dayRegex = /^> \[!card\] (.+)$/; const taskRegex = /^> - \[( |x|X)\] ?(.*)$/; const rows = []; let currentDay = null; let planned = 0; let done = 0; let total = 0; let count = 0; function flush() { if (currentDay && planned > 0) { const eff = (done / planned) * 100; rows.push([ currentDay, `${done}/${planned}`, eff.toFixed(1) + '%' ]); total += eff; count++; } } for (const line of lines) { const dm = line.match(dayRegex); if (dm) { flush(); currentDay = dm[1]; planned = 0; done = 0; continue; } const tm = line.match(taskRegex); if (tm && tm[2].trim() !== '') { planned++; if (tm[1].toLowerCase() === 'x') { done++; } } } flush(); dv.table( ['Day', 'Completed', 'Efficiency'], rows ); if (count > 0) { dv.paragraph( `**Month average efficiency: ${(total / count).toFixed(1)}%** (based on ${count} planned days)` ); } else { dv.paragraph('No tasks planned yet.'); }
```
