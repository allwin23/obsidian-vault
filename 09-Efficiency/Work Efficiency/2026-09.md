``` dataviewjs
const { MarkdownView } = require("obsidian");

const container = dv.el("div", "", { cls: "eff-day-jump" });

const label = document.createElement("label");
label.textContent = "Jump to a day: ";

const input = document.createElement("input");
input.type = "date";

const btn = document.createElement("button");
btn.textContent = "Go";

async function jump() {
    if (!input.value) return;

    const [year, month, day] = input.value.split("-").map(Number);
    const date = new Date(year, month - 1, day);
    const target = `${day} ${date.toLocaleDateString("en-US", { month: "long" })}`;

    const view = app.workspace.getActiveViewOfType(MarkdownView);
    if (!view) return;

    const file = app.vault.getAbstractFileByPath(dv.current().file.path);
    if (!file) return;

    const content = await app.vault.read(file);
    const lines = content.split("\n");

    const targetLine = lines.findIndex(line =>
        line.startsWith("> [!card]") && line.includes(target)
    );

    if (targetLine < 0) {
        new Notice(`No day found for ${target}.`);
        return;
    }

    if (view.editor) {
        view.editor.setCursor({ line: targetLine, ch: 0 });
        view.editor.scrollIntoView({
            from: { line: targetLine, ch: 0 },
            to: { line: targetLine, ch: 0 }
        }, true);
    } else {
        const callouts = view.containerEl.querySelectorAll(".callout");
        for (const callout of callouts) {
            const title = callout.querySelector(".callout-title");
            if (title && title.textContent.includes(target)) {
                callout.scrollIntoView({ behavior: "smooth", block: "start" });
                return;
            }
        }
    }
}

btn.addEventListener("click", jump);
input.addEventListener("keydown", event => {
    if (event.key === "Enter") jump();
});

container.appendChild(label);
container.appendChild(input);
container.appendChild(btn);
```

## Month Summary

``` dataviewjs
const file = dv.current().file;
const abstractFile = app.vault.getAbstractFileByPath(file.path);

if (!abstractFile) {
    dv.paragraph("Unable to locate this note.");
} else {
    const content = await app.vault.read(abstractFile);
    const lines = content.split("\n");

    const dayRegex = /^>\s*\[!card\]\s*(.+)$/;
    const taskRegex = /^>\s*-\s*\[( |x|X)\]\s*(.*)$/;

    const rows = [];

    let currentDay = null;
    let planned = 0;
    let done = 0;
    let totalEfficiency = 0;
    let plannedDays = 0;

    function flushDay() {
        if (currentDay !== null && planned > 0) {
            const efficiency = (done / planned) * 100;

            rows.push([
                currentDay,
                `${done}/${planned}`,
                `${efficiency.toFixed(1)}%`
            ]);

            totalEfficiency += efficiency;
            plannedDays++;
        }
    }

    for (const line of lines) {
        const dayMatch = line.match(dayRegex);

        if (dayMatch) {
            flushDay();

            currentDay = dayMatch[1].trim();
            planned = 0;
            done = 0;

            continue;
        }

        const taskMatch = line.match(taskRegex);

        if (taskMatch && taskMatch[2].trim() !== "") {
            planned++;

            if (taskMatch[1].toLowerCase() === "x") {
                done++;
            }
        }
    }

    flushDay();

    if (plannedDays > 0) {
        dv.table(["Day", "Completed", "Efficiency"], rows);

        dv.paragraph(
            `**Month average efficiency: ${(totalEfficiency / plannedDays).toFixed(1)}%** ` +
            `(based on ${plannedDays} planned days)`
        );
    } else {
        dv.paragraph("No tasks planned yet.");
    }
}
```
