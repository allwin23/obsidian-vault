Plan each day's tasks at the start of the day. Efficiency is calculated automatically as completed ÷ planned tasks.

```dataviewjs
const { MarkdownView } = require('obsidian');

const container = dv.el('div', '', { cls: 'eff-day-jump' });
const label = document.createElement('label');
label.textContent = 'Jump to a day: ';
const input = document.createElement('input');
input.type = 'date';
const btn = document.createElement('button');
btn.textContent = 'Go';

async function jump() {
  if (!input.value) return;
  const d = new Date(input.value + 'T00:00:00');
  const target = `${d.getDate()} ${d.toLocaleDateString('default', { month: 'long' })}`;

  const file = app.vault.getAbstractFileByPath(dv.current().file.path);
  const content = await app.vault.read(file);
  const lines = content.split('\n');
  let targetLine = -1;
  for (let i = 0; i < lines.length; i++) {
    if (lines[i].startsWith('> [!card]') && lines[i].includes(target)) {
      targetLine = i;
      break;
    }
  }
  if (targetLine === -1) return;

  const view = app.workspace.getActiveViewOfType(MarkdownView);
  if (!view) return;

  if (view.getMode && view.getMode() === 'preview' && view.previewMode?.applyScroll) {
    view.previewMode.applyScroll(targetLine);
  } else if (view.editor) {
    view.editor.setCursor({ line: targetLine, ch: 0 });
    view.editor.scrollIntoView({ from: { line: targetLine, ch: 0 }, to: { line: targetLine, ch: 0 } }, true);
  }
}

btn.addEventListener('click', jump);
input.addEventListener('keydown', (e) => { if (e.key === 'Enter') jump(); });

container.appendChild(label);
container.appendChild(input);
container.appendChild(btn);
```

## Month Summary

```dataviewjs
const file = dv.current();
const cache = app.metadataCache.getFileCache(app.vault.getAbstractFileByPath(file.file.path));
const content = await app.vault.read(app.vault.getAbstractFileByPath(file.file.path));
const lines = content.split('\n');
const dayRegex = /^> \[!card\] (.+)$/;
const taskRegex = /^> - \[( |x|X)\] ?(.*)$/;
const rows = [];
let currentDay = null, planned = 0, done = 0;
let total = 0, count = 0;
function flush() {
  if (currentDay && planned > 0) {
    const eff = (done / planned) * 100;
    rows.push([currentDay, `${done}/${planned}`, eff.toFixed(1) + '%']);
    total += eff; count++;
  }
}
for (const line of lines) {
  const dm = line.match(dayRegex);
  if (dm) { flush(); currentDay = dm[1]; planned = 0; done = 0; continue; }
  const tm = line.match(taskRegex);
  if (tm && tm[2].trim() !== '') {
    planned++;
    if (tm[1].toLowerCase() === 'x') done++;
  }
}
flush();
dv.table(['Day','Completed','Efficiency'], rows);
if (count > 0) {
  dv.paragraph(`**Month average efficiency: ${(total/count).toFixed(1)}%** (based on ${count} planned days)`);
} else {
  dv.paragraph('No tasks planned yet.');
}
```
