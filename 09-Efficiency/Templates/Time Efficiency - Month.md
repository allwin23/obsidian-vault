## Month Summary

```dataviewjs
const p = dv.current();
const rows = [];
let reqTotal = 0, reqCount = 0, actTotal = 0, actCount = 0;
for (let d = 1; d <= 30; d++) {
  const dd = String(d).padStart(2, '0');
  const planned = p['planned_' + dd];
  const worked = p['worked_' + dd];
  if (worked === undefined || worked === null || worked === '') continue;
  const req = (worked / 12) * 100;
  reqTotal += req; reqCount++;
  let act = null;
  if (planned !== undefined && planned !== null && planned !== '' && planned > 0) {
    act = (worked / planned) * 100;
    actTotal += act; actCount++;
  }
  rows.push([dd, planned ?? '—', worked, req.toFixed(1) + '%', act !== null ? act.toFixed(1) + '%' : '—']);
}
dv.table(['Day','Planned','Worked','Required Eff.','Actual Eff.'], rows);
if (reqCount > 0) {
  dv.paragraph(`**Month avg — Required: ${(reqTotal/reqCount).toFixed(1)}%${actCount ? ' · Actual: ' + (actTotal/actCount).toFixed(1) + '%' : ''}** (based on ${reqCount} logged days)`);
} else {
  dv.paragraph('No days logged yet.');
}
```
